// Usage: SCRATCH=<dir> CHROME_PATH=... bun upgrade-probe.ts <unpacked-0.2.1> <unpacked-0.3.0> <shots-dir>
// Runs the released 0.2.1 from a fixed directory in a profile that restores its last session, links a
// Chrome group to Linklog on a fake production page, quits, overwrites the directory with 0.3.0 and
// relaunches the same profile from a second directory (both manifests carry the same `key`, so the same
// extension id): Chrome updates the extension and restores the tabs and group.
import { chromium, type BrowserContext, type Page } from "playwright-core";
import { mkdtempSync, rmSync } from "node:fs";
import { join } from "node:path";
const [oldExt, newExt, shots] = process.argv.slice(2) as [string, string, string];
let dir = oldExt;
const profile = mkdtempSync(join(process.env.SCRATCH!, "prof-"));
const wait = (ms: number) => new Promise((r) => setTimeout(r, ms));
const requests: string[] = [];

async function open(): Promise<BrowserContext> {
  const ctx = await chromium.launchPersistentContext(profile, {
    headless: false, executablePath: process.env.CHROME_PATH, viewport: { width: 1280, height: 800 },
    args: [`--disable-extensions-except=${dir}`, `--load-extension=${dir}`, "--restore-last-session"],
  });
  ctx.on("request", (r) => { if (r.frame()?.url().startsWith("chrome-extension://") || r.serviceWorker()) requests.push(r.url()); });
  await ctx.route("https://linklog.facundopri.com/**", (route) =>
    route.fulfill({ contentType: "text/html", body: `<!doctype html><html><head><meta name="linklog-app" content="1"><title>Linklog (fake)</title></head><body>fake linklog</body></html>` }));
  await ctx.route("https://secret.example/**", (route) => route.fulfill({ contentType: "text/html", body: `<title>Secret page title</title>secret` }));
  return ctx;
}
const ask = (page: Page, command: string, args: object = {}) =>
  page.evaluate(({ command, args }) => new Promise<any>((resolve) => {
    const id = crypto.randomUUID();
    const t = setTimeout(() => resolve({ timeout: true }), 6000);
    addEventListener("message", function on(e) {
      const r = (e.data as any)?.response;
      if ((e.data as any)?.channel === "linklog-bridge" && r?.id === id) { clearTimeout(t); removeEventListener("message", on); resolve(r); }
    });
    postMessage({ channel: "linklog-bridge", from: "page", request: { id, command, args } }, location.origin);
  }), { command, args });
const bridged = (p: Page) => p.waitForFunction(() => document.documentElement.dataset.linklogBridge !== undefined, null, { timeout: 8000 });
const scope = "local:probe";

let ctx = await open();
const sw = ctx.serviceWorkers()[0] ?? (await ctx.waitForEvent("serviceworker"));
const id = new URL(sw.url()).host;
await wait(1500);
const secret = await ctx.newPage(); await secret.goto("https://secret.example/path?q=private");
let ll = await ctx.newPage(); await ll.goto("https://linklog.facundopri.com/browser"); await bridged(ll);
const oldHello = await ask(ll, "hello", { libraryScope: scope });
const groupId = await sw.evaluate(async () => {
  const [tab] = await chrome.tabs.query({ url: "https://secret.example/*" });
  const g = await chrome.tabs.group({ tabIds: [tab!.id!] });
  await chrome.tabGroups.update(g, { title: "Research", color: "blue" });
  return g;
});
const link = await ask(ll, "updateGroup", { groupId, pinnedId: "pinned-1" });
await wait(1500);
const oldLocal = await sw.evaluate(() => chrome.storage.local.get(null));
for (const p of ctx.pages()) if (p.url().startsWith("chrome-extension://")) await p.close();
await wait(500);
await ctx.close();

dir = newExt;
// A store update re-registers the service worker of the new version; a command-line reload keeps the
// old registration, so drop it (only the worker's script cache, not storage).
rmSync(join(profile, "Default", "Service Worker"), { recursive: true, force: true });
ctx = await open();
await wait(4000);
const restoredTabs = ctx.pages().map((p) => p.url());
const connectTabs = restoredTabs.filter((u) => u.endsWith("/connect.html")).length;
// The popup reads storage (it writes nothing about tabs).
const popup = await ctx.newPage(); await popup.setViewportSize({ width: 300, height: 220 });
await popup.goto(`chrome-extension://${id}/popup.html`); await wait(800);
const storage = () => popup.evaluate(async () => ({ session: await chrome.storage.session.get(null), local: await chrome.storage.local.get(null) }));
const afterUpdate = { version: await popup.evaluate(() => chrome.runtime.getManifest().version), connectTabs, restoredTabs, storage: await storage() };
await popup.screenshot({ path: join(shots, "upgrade-popup-before-consent.png") });
const popupText = await popup.evaluate(() => document.body.innerText);
ll = ctx.pages().find((p) => p.url().startsWith("https://linklog.facundopri.com/")) ?? (await ctx.newPage());
if (!ll.url().startsWith("https://linklog")) await ll.goto("https://linklog.facundopri.com/browser");
else await ll.reload();
await bridged(ll);
const helloBefore = await ask(ll, "hello", { libraryScope: scope });
const storageBeforeAccept = await storage();

const connect = await ctx.newPage(); await connect.goto(`chrome-extension://${id}/connect.html`);
await connect.click("#accept");
await wait(3000);
const helloAfter = await ask(ll, "hello", { libraryScope: scope });
const snap = await ask(ll, "snapshot");
const after = await storage();
console.log(JSON.stringify({
  oldVersion: oldHello?.result?.extensionVersion, link, oldLocal,
  afterUpdate, popupText, helloBefore, storageBeforeAccept,
  helloAfter: { ok: helloAfter.ok, version: helloAfter.result?.extensionVersion, installationId: helloAfter.result?.installationId },
  groupsAfter: snap.result?.groups, localKeysAfter: Object.keys(after.local),
  extensionNetworkRequests: requests.filter((u) => !u.startsWith("chrome-extension://")),
}, null, 1));
await ctx.close();
